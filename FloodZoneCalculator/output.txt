902 '        private void DrawFlowChart(Graphics graphics, Rectangle bounds, bool logarithmic)\n'
903 '        {\n'
904 '            graphics.Clear(ModernTheme.Background);\n'
905 '            var points = ReadFlowPoints();\n'
906 '            DrawAxes(graphics, bounds, "Расход, м³/с", logarithmic ? "Обеспеченность, % (лог.)" : "Обеспеченность, %");\n'
907 '            if (points.Count < 2) return;\n'
908 '\n'
909 '            var plot = GetPlotRectangle(bounds);\n'
910 '            var minX = logarithmic ? Math.Max(0.01, points.Min(p => p.P)) : 0;\n'
911 '            var maxX = 100d;\n'
912 '            var maxY = Math.Max(1, points.Max(p => p.Q));\n'
913 '            PointF Map(FlowPoint p)\n'
914 '            {\n'
915 '                var x = logarithmic\n'
916 '                    ? (Math.Log10(Math.Max(minX, p.P)) - Math.Log10(minX)) /\n'
917 '                      (Math.Log10(maxX) - Math.Log10(minX))\n'
918 '                    : p.P / maxX;\n'
919 '                return new PointF((float)(plot.Left + x * plot.Width),\n'
920 '                    (float)(plot.Bottom - p.Q / maxY * plot.Height));\n'
921 '            }\n'
922 '\n'
923 '            using var pen = new Pen(Color.FromArgb(37, 99, 235), 2);\n'
924 '            var mapped = points.Select(Map).ToArray();\n'
925 '            graphics.DrawLines(pen, mapped);\n'
926 '            using var brush = new SolidBrush(Color.FromArgb(37, 99, 235));\n'
927 '            foreach (var point in mapped)\n'
928 '                graphics.FillEllipse(brush, point.X - 3, point.Y - 3, 6, 6);\n'
929 '        }\n'
